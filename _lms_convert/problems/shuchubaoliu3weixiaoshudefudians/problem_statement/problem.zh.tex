\problemname{输出保留3位小数的浮点数}

\subsection*{题目描述}
读入一个浮点数，保留 \texttt{3} 位小数输出这个浮点数。

\subsection*{输入格式}
输入一个浮点数 \texttt{shu}。

\subsection*{输出格式}
输出这个浮点数，保留 \texttt{3} 位小数。

\subsection*{样例输入}
\begin{verbatim}
3.14159
\end{verbatim}

\subsection*{样例输出}
\begin{verbatim}
3.142
\end{verbatim}
